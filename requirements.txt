# MCP (Model Context Protocol) Dependencies
# Anthropic's Model Context Protocol client library — lets the MCP Agent page
# connect to and call tools on external MCP servers (e.g. Zapier, Notion).
# `mcp` is pinned because langchain-mcp-adapters==0.2.2 does not cap it, and
# mcp>=2.0.0 renamed internals (RequestContext moved out of mcp.shared.context),
# which breaks langchain_mcp_adapters.client on import ("cannot import name
# 'RequestContext' from 'mcp.shared.context'"). Pin stays until adapters is upgraded.
# LangChain's adapter layer that turns an MCP server's tools into LangChain
# tools the agent can call — the bridge between MCP and create_agent().
langchain-mcp-adapters==0.2.2
mcp==1.29.0

# Used for simple HTTP calls where a full LangChain/MCP client isn't needed.

# Research agent tools

# Async support for Streamlit + MCP

nest_asyncio==1.6.0

# =============================================================================
# Cross-exercise compatibility pins (not used by this project's own code)
# =============================================================================
# If this environment is shared with other bootcamp exercises, older installs of
# these three packages conflict with the LangChain v1 stack above:
#   - langchain-experimental 0.3.4 needs langchain-community<0.4.0 (this project pins 0.4.2)
#   - openai-agents 0.2.10 needs openai<2 (this project resolves to openai>=2.4x via langchain-openai/mcp)
#   - gradio 5.44.1 needs pydantic<2.12, which conflicts with openai-agents needing pydantic>=2.12.2
# Pinning them here to versions verified compatible with everything above (tested with
# `pip check` -> "No broken requirements found"). Remove this block if your environment
# doesn't have these installed for another exercise.
langchain-experimental==0.4.2
openai-agents==0.20.0
gradio==6.25.0

# chromadb 1.0.12 hard-pins fastapi==0.115.9, which needs an old starlette (<0.46) that
# cannot coexist with the newer starlette gradio==6.25.0 above requires (>=1.0.1) - no
# fastapi version satisfies both, so this can only be fixed on the chromadb side. From
# 1.1.0 onward chromadb only requires fastapi under its optional "dev" extra (not part of
# a normal install), which removes the conflict entirely.
chromadb==1.1.1
